  <oap:target id="views-section">
    CREATE OR REPLACE VIEW <oap:target id="order-summary-view-name">order_summary</oap:target> AS
    SELECT
      <oap:target id="order-summary-order-id">o.id</oap:target> AS <oap:target id="order-summary-order-id-alias">order_id</oap:target>,
      <oap:target id="order-summary-order-number">o.order_number</oap:target> AS <oap:target id="order-summary-order-number-alias">order_number</oap:target>,
      <oap:target id="order-summary-user-id">o.user_id</oap:target>,
      <oap:target id="order-summary-user-name">u.first_name || ' ' || u.last_name</oap:target> AS <oap:target id="order-summary-user-name-alias">customer_name</oap:target>,
      <oap:target id="order-summary-status">o.status</oap:target>,
      <oap:target id="order-summary-subtotal">o.subtotal</oap:target>,
      <oap:target id="order-summary-tax">o.tax</oap:target>,
      <oap:target id="order-summary-shipping">o.shipping</oap:target>,
      <oap:target id="order-summary-discount">o.discount</oap:target>,
      <oap:target id="order-summary-total">o.total</oap:target>,
      <oap:target id="order-summary-item-count">COUNT(oi.id)</oap:target> AS <oap:target id="order-summary-item-count-alias">item_count</oap:target>,
      <oap:target id="order-summary-created-at">o.placed_at</oap:target> AS <oap:target id="order-summary-created-at-alias">placed_at</oap:target>
    FROM <oap:target id="order-summary-orders-table">orders</oap:target> o
    JOIN <oap:target id="order-summary-users-table">users</oap:target> u ON u.id = o.user_id
    LEFT JOIN <oap:target id="order-summary-order-items-table">order_items</oap:target> oi ON oi.order_id = o.id
    GROUP BY o.id, u.first_name, u.last_name;

    CREATE OR REPLACE VIEW <oap:target id="product-ratings-view-name">product_ratings</oap:target> AS
    SELECT
      <oap:target id="product-ratings-product-id">p.id</oap:target> AS <oap:target id="product-ratings-product-id-alias">product_id</oap:target>,
      <oap:target id="product-ratings-product-name">p.name</oap:target> AS <oap:target id="product-ratings-product-name-alias">product_name</oap:target>,
      <oap:target id="product-ratings-average-rating">COALESCE(AVG(r.rating), 0)</oap:target> AS <oap:target id="product-ratings-average-rating-alias">average_rating</oap:target>,
      <oap:target id="product-ratings-rating-count">COUNT(r.id)</oap:target> AS <oap:target id="product-ratings-rating-count-alias">rating_count</oap:target>
    FROM <oap:target id="product-ratings-products-table">products</oap:target> p
    LEFT JOIN <oap:target id="product-ratings-reviews-table">reviews</oap:target> r ON r.product_id = p.id
    GROUP BY p.id, p.name;

    CREATE OR REPLACE VIEW <oap:target id="low-stock-products-view-name">low_stock_products</oap:target> AS
    SELECT
      <oap:target id="low-stock-product-id">p.id</oap:target> AS <oap:target id="low-stock-product-id-alias">product_id</oap:target>,
      <oap:target id="low-stock-product-name">p.name</oap:target> AS <oap:target id="low-stock-product-name-alias">product_name</oap:target>,
      <oap:target id="low-stock-sku">p.sku</oap:target>,
      <oap:target id="low-stock-on-hand">i.quantity_on_hand</oap:target>,
      <oap:target id="low-stock-reserved">i.reserved_quantity</oap:target>,
      <oap:target id="low-stock-reorder-level">i.reorder_level</oap:target>
    FROM <oap:target id="low-stock-products-table">products</oap:target> p
    JOIN <oap:target id="low-stock-inventory-table">inventory</oap:target> i ON i.product_id = p.id
    WHERE <oap:target id="low-stock-condition">i.quantity_on_hand &lt;= i.reorder_level</oap:target>;

    CREATE MATERIALIZED VIEW <oap:target id="monthly-sales-summary-view-name">monthly_sales_summary</oap:target> AS
    SELECT
      EXTRACT(YEAR FROM o.placed_at)::INT AS <oap:target id="monthly-sales-summary-year-alias">year</oap:target>,
      EXTRACT(MONTH FROM o.placed_at)::INT AS <oap:target id="monthly-sales-summary-month-alias">month</oap:target>,
      SUM(o.total) AS <oap:target id="monthly-sales-summary-total-revenue-alias">total_revenue</oap:target>,
      COUNT(o.id) AS <oap:target id="monthly-sales-summary-order-count-alias">order_count</oap:target>,
      COALESCE(AVG(o.total), 0) AS <oap:target id="monthly-sales-summary-average-order-value-alias">average_order_value</oap:target>
    FROM <oap:target id="monthly-sales-summary-orders-table">orders</oap:target> o
    GROUP BY EXTRACT(YEAR FROM o.placed_at), EXTRACT(MONTH FROM o.placed_at);
  </oap:target></oap:target>
